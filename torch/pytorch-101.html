PyTorch 101

For people who already know the math — this is the API map.

Tensors

Tensors are the fundamental data structure — N-dimensional arrays with dtype, device, and a gradient tape attached.

Creation

import torch

# From data
x = torch.tensor([1.0, 2.0, 3.0])                   # infers dtype
x = torch.tensor([1.0, 2.0], dtype=torch.float32)   # explicit dtype

# Shape-based
torch.zeros(3, 4)       # 3x4 of zeros (float32)
torch.ones(3, 4)        # 3x4 of ones
torch.rand(3, 4)        # uniform [0,1)
torch.randn(3, 4)       # standard normal N(0,1)
torch.arange(0, 10, 2)  # [0, 2, 4, 6, 8]

# From numpy (shares memory!)
x = torch.from_numpy(arr)
arr = x.numpy()          # only works on CPU tensors

Shape manipulation

x = torch.randn(8, 16)

x.shape               # torch.Size([8, 16])
x.reshape(2, 4, 16)   # new shape, same data
x.view(2, 4, 16)      # like reshape but requires contiguous memory
x.permute(1, 0)       # transpose: (16, 8)
x.unsqueeze(0)        # add dim at pos 0 → (1, 8, 16)
x.squeeze()           # remove all size-1 dims
x[0]                  # index → shape (16,)
x[:, 3:7]             # slice → (8, 4)
reshape vs view Both change shape without copying data. view requires the tensor to be contiguous in memory; if it's not (e.g. after permute), call .contiguous() first or just use reshape which handles this automatically.

Common ops

a = torch.randn(4, 4)
b = torch.randn(4, 4)

a @ b              # matrix multiply (same as torch.matmul)
a * b              # element-wise multiply
torch.sum(a, dim=0)   # sum along rows → shape (4,)
torch.mean(a, dim=1)  # mean along cols → shape (4,)
torch.softmax(a, dim=-1)  # softmax along last dim
torch.cat([a, b], dim=0)  # concat → (8, 4)
torch.stack([a, b], dim=0) # new dim → (2, 4, 4)

Autograd

PyTorch builds a computation graph dynamically as you run forward operations. .backward() traverses it in reverse to compute gradients via chain rule.

x = torch.tensor([2.0], requires_grad=True)
y = x ** 2 + 3 * x   # builds graph
y.backward()          # dy/dx = 2x + 3 = 7
x.grad                 # tensor([7.])
requires_grad propagates Any tensor computed from a requires_grad=True tensor also has requires_grad=True. The model parameters are leaves with this flag set — everything else in the forward pass inherits it.

Detaching from the graph

# Don't track gradients for this operation
with torch.no_grad():
    y = model(x)   # no graph built → faster, less memory

# Detach a tensor from the graph (e.g. for logging)
loss_val = loss.detach().item()   # scalar Python float

# Stop gradients flowing through a specific path
target = output.detach()   # common in policy gradient / target networks
Gradient accumulation gotcha Gradients accumulate by default — they don't reset each step. You must call optimizer.zero_grad() before each backward pass, or they'll keep adding up across batches.

nn.Module

The base class for all models. Subclass it, define layers in __init__, describe the forward pass in forward.

import torch.nn as nn

class MLP(nn.Module):
    def __init__(self, in_dim, hidden, out_dim):
        super().__init__()
        self.layers = nn.Sequential(
            nn.Linear(in_dim, hidden),
            nn.ReLU(),
            nn.Linear(hidden, out_dim)
        )

    def forward(self, x):
        return self.layers(x)

model = MLP(784, 256, 10)
out = model(x)   # calls forward() — don't call .forward() directly

Building Networks

Three patterns for composing layers — each with different tradeoffs for readability and flexibility.

Pattern 1: nn.Sequential — linear stacks

# Good for: simple feedforward, conv blocks with no branching
encoder = nn.Sequential(
    nn.Linear(784, 512),
    nn.BatchNorm1d(512),
    nn.ReLU(),
    nn.Dropout(0.3),
    nn.Linear(512, 128),
    nn.ReLU(),
)

# Can also build dynamically with OrderedDict for named layers
from collections import OrderedDict
block = nn.Sequential(OrderedDict([
    ('fc1',   nn.Linear(128, 64)),
    ('relu1', nn.ReLU()),
    ('fc2',   nn.Linear(64, 10)),
]))

Pattern 2: nn.ModuleList — dynamic / repeated blocks

# Good for: variable depth, iterating over layers in forward()
class DeepNet(nn.Module):
    def __init__(self, widths):  # e.g. [784, 512, 256, 128, 10]
        super().__init__()
        self.layers = nn.ModuleList([
            nn.Linear(widths[i], widths[i+1])
            for i in range(len(widths) - 1)
        ])

    def forward(self, x):
        for i, layer in enumerate(self.layers):
            x = layer(x)
            if i < len(self.layers) - 1:   # no activation on last layer
                x = torch.nn.functional.relu(x)
        return x
Always use ModuleList / ModuleDict — never plain Python lists If you store submodules in a plain Python list, PyTorch can't see them. They won't appear in parameters(), won't get moved by .to(device), and won't be saved in state_dict(). Always wrap in nn.ModuleList or nn.ModuleDict.

Pattern 3: Full Module — branches, skip connections, multiple inputs

# Good for: ResNet blocks, encoder-decoder, multi-head models
class ResidualBlock(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.block = nn.Sequential(
            nn.Linear(dim, dim),
            nn.LayerNorm(dim),
            nn.ReLU(),
            nn.Linear(dim, dim),
            nn.LayerNorm(dim),
        )

    def forward(self, x):
        return torch.nn.functional.relu(x + self.block(x))   # skip connection


class EncoderDecoder(nn.Module):
    def __init__(self):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(784, 256), nn.ReLU())
        self.decoder = nn.Sequential(nn.Linear(256, 784), nn.Sigmoid())

    def forward(self, x):
        z = self.encoder(x)   # latent
        return self.decoder(z), z   # return both for loss computation

CNN building block

class ConvBlock(nn.Module):
    """Conv → BN → ReLU — the fundamental CNN unit"""
    def __init__(self, in_ch, out_ch, kernel=3, stride=1):
        super().__init__()
        self.block = nn.Sequential(
            nn.Conv2d(in_ch, out_ch, kernel, stride, padding=kernel//2, bias=False),
            nn.BatchNorm2d(out_ch),   # bias=False because BN has its own bias
            nn.ReLU(inplace=True),
        )

    def forward(self, x):
        return self.block(x)


# Stacking into a small CNN
class SmallCNN(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.features = nn.Sequential(
            ConvBlock(3, 32),    # (B,3,H,W) → (B,32,H,W)
            nn.MaxPool2d(2),     # → (B,32,H/2,W/2)
            ConvBlock(32, 64),
            nn.MaxPool2d(2),     # → (B,64,H/4,W/4)
            ConvBlock(64, 128),
            nn.AdaptiveAvgPool2d((1, 1)),  # → (B,128,1,1) — any input size works
        )
        self.classifier = nn.Linear(128, num_classes)

    def forward(self, x):
        x = self.features(x)
        x = x.flatten(1)          # (B,128,1,1) → (B,128)
        return self.classifier(x)

Activations

The choice of activation is an architectural decision — it affects gradient flow, expressivity, and training stability.

ActivationFormulaUse when
nn.ReLUmax(0, x)Default for CNNs and MLPs. Fast, sparse.
nn.GELUx·Φ(x)Transformers (BERT, GPT). Smoother than ReLU.
nn.SiLU / Swishx·σ(x)Modern CNNs (EfficientNet, YOLOv8). Non-monotonic.
nn.LeakyReLUmax(αx, x)When dying ReLU is a problem. α typically 0.01–0.2.
nn.Tanhtanh(x)RNNs, outputs bounded to (-1,1).
nn.Sigmoidσ(x)Binary classification output only. Vanishes in hidden layers.
nn.Softmaxeˣᵢ / ΣeˣMulti-class output. Use dim=-1. Built into CrossEntropyLoss.
import torch.nn.functional as F

# Module form (use inside __init__ for stored layers)
self.act = nn.GELU()

# Functional form (use inside forward() for stateless ops)
x = F.relu(x)
x = F.gelu(x)
x = F.softmax(x, dim=-1)
x = F.sigmoid(x)

# inplace=True saves memory but breaks autograd in some cases — use with care
x = F.relu(x, inplace=True)
Don't apply Softmax before CrossEntropyLoss nn.CrossEntropyLoss internally applies log-softmax + NLL. If you apply softmax yourself first, you're passing probabilities to a function expecting raw logits — loss will be wrong and training will break silently.

Normalization

Normalization layers stabilize training by controlling the scale and distribution of activations. Which one to use depends on your architecture.

LayerNormalizes overUse when
BatchNorm1d/2dBatch dimension (N)CNNs with large batches. Bad with small batches.
LayerNormFeature dimension (C)Transformers, RNNs. Works with batch_size=1.
GroupNormGroups of channelsCNNs with small batches (detection, segmentation).
InstanceNormEach sample independentlyStyle transfer, generative models.
# BatchNorm — learns gamma (scale) and beta (shift)
nn.BatchNorm1d(num_features)   # for (B, F) inputs
nn.BatchNorm2d(num_channels)   # for (B, C, H, W) inputs

# LayerNorm — normalizes last N dims
nn.LayerNorm(hidden_size)            # normalize over embedding dim
nn.LayerNorm([seq_len, hidden_size]) # normalize over both dims

# GroupNorm — num_groups divides num_channels evenly
nn.GroupNorm(num_groups=32, num_channels=256)

# Where to place norm — two conventions:
# Post-norm (original): Linear → Activation → Norm  (classic transformers)
# Pre-norm (modern):   Norm → Linear → Activation   (more stable training)
BatchNorm train vs eval behavior During training, BN normalizes using the current batch's mean/variance and updates running statistics. During eval(), it uses those frozen running stats instead. This is why forgetting model.eval() gives inconsistent validation metrics — BN is still operating in batch mode.

Regularization

Dropout

# Standard dropout — zeroes random neurons with probability p
nn.Dropout(p=0.5)        # for Linear layers
nn.Dropout2d(p=0.1)      # for Conv layers — drops entire channels
nn.AlphaDropout(p=0.1)   # for SELU activations — preserves self-normalizing property

# Dropout is ONLY active during training mode
# model.eval() disables it automatically — no manual toggle needed

# Functional form for use inside forward()
x = F.dropout(x, p=0.5, training=self.training)  # self.training is the mode flag

Weight decay (L2 regularization)

# Applied via optimizer, not as a layer
optimizer = torch.optim.AdamW(model.parameters(), lr=1e-3, weight_decay=0.01)

# Apply weight decay only to weights, not biases or norms
decay_params    = [p for n, p in model.named_parameters() if 'bias' not in n]
no_decay_params = [p for n, p in model.named_parameters() if 'bias' in n]
optimizer = torch.optim.AdamW([
    {'params': decay_params,    'weight_decay': 0.01},
    {'params': no_decay_params, 'weight_decay': 0.0},
], lr=1e-3)

Weight Initialization

PyTorch has sensible defaults (Kaiming uniform for Linear/Conv, uniform for Embedding), but explicit init is important for custom architectures.

import torch.nn.init as init

# Common initializers
init.kaiming_normal_(layer.weight, mode='fan_out', nonlinearity='relu')  # He init
init.xavier_uniform_(layer.weight)    # Glorot init — for tanh/sigmoid
init.normal_(layer.weight, mean=0, std=0.02)   # GPT-style small init
init.zeros_(layer.bias)
init.ones_(layer.weight)              # for LayerNorm gamma

# Apply to an entire model with apply()
def init_weights(module):
    if isinstance(module, nn.Linear):
        init.kaiming_normal_(module.weight)
        if module.bias is not None:
            init.zeros_(module.bias)
    elif isinstance(module, nn.Embedding):
        init.normal_(module.weight, std=0.02)
    elif isinstance(module, (nn.BatchNorm1d, nn.LayerNorm)):
        init.ones_(module.weight)
        init.zeros_(module.bias)

model.apply(init_weights)   # walks every submodule recursively
Why Kaiming for ReLU, Xavier for Tanh? Both try to keep activation variance stable across layers. Kaiming (He) accounts for ReLU's dead half — it scales by √(2/fan_in). Xavier assumes symmetric activations — it scales by √(2/(fan_in + fan_out)). Using Xavier with ReLU causes vanishing activations in deep networks.

Custom Layers & Functional API

Custom layer with learnable parameters

class ScaledDotProductAttention(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.scale = dim ** -0.5       # 1/√d_k
        self.q = nn.Linear(dim, dim)
        self.k = nn.Linear(dim, dim)
        self.v = nn.Linear(dim, dim)

    def forward(self, x):
        Q = self.q(x)
        K = self.k(x)
        V = self.v(x)
        scores = (Q @ K.transpose(-2, -1)) * self.scale   # (B, T, T)
        attn   = F.softmax(scores, dim=-1)
        return attn @ V


class LearnedPositionalEncoding(nn.Module):
    def __init__(self, max_len, dim):
        super().__init__()
        self.pe = nn.Embedding(max_len, dim)   # learned, not sinusoidal

    def forward(self, x):
        positions = torch.arange(x.size(1), device=x.device)
        return x + self.pe(positions)   # broadcast over batch dim

nn.Parameter — raw learnable tensors

class LayerScale(nn.Module):
    """Learnable per-channel scale factor (used in DeiT, CaiT)"""
    def __init__(self, dim, init_value=1e-4):
        super().__init__()
        # nn.Parameter registers tensor as a learnable parameter
        self.gamma = nn.Parameter(torch.ones(dim) * init_value)

    def forward(self, x):
        return x * self.gamma   # element-wise scale

Data Pipeline

PyTorch's data loading is a two-class system: Dataset defines what data exists, DataLoader defines how to iterate over it.

from torch.utils.data import Dataset, DataLoader

class MyDataset(Dataset):
    def __init__(self, X, y):
        self.X = X  # tensors
        self.y = y

    def __len__(self):
        return len(self.X)

    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]   # (input, label) pair

dataset = MyDataset(X_train, y_train)
loader  = DataLoader(dataset, batch_size=32, shuffle=True, num_workers=4)

for X_batch, y_batch in loader:
    # X_batch.shape → (32, ...)
    ...
num_workers Spawns background processes that prefetch data in parallel so the GPU is never idle waiting for I/O. Rule of thumb: 4 per GPU. Set to 0 when debugging (avoids multiprocessing overhead and confusing tracebacks).

Training Loop

The canonical pattern. Every step is explicit — this is by design.

model = MLP(784, 256, 10).to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
criterion = nn.CrossEntropyLoss()

for epoch in range(num_epochs):
    model.train()                         # ← sets training mode
    for X, y in train_loader:
        X, y = X.to(device), y.to(device)

        optimizer.zero_grad()             # ① reset gradients
        logits = model(X)                 # ② forward pass
        loss   = criterion(logits, y)     # ③ compute loss
        loss.backward()                   # ④ backprop
        optimizer.step()                  # ⑤ update weights
Why in this exact order? zero_grad → forward → loss → backward → step. Deviating breaks things: if you call backward() twice without zero_grad() gradients add; if you call step() before backward() you update on stale gradients.

Gradient clipping

# Insert between backward() and step()
nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)

Prevents exploding gradients — essential for RNNs and Transformers. Clips the global norm of all parameter gradients together.

Optimizers & LR Scheduling

# Common optimizers
torch.optim.SGD(model.parameters(), lr=0.01, momentum=0.9)
torch.optim.Adam(model.parameters(), lr=1e-3, weight_decay=1e-4)
torch.optim.AdamW(model.parameters(), lr=1e-3, weight_decay=0.01)

# LR Schedulers (call scheduler.step() after each epoch)
scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=10, gamma=0.1)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=50)
scheduler = torch.optim.lr_scheduler.OneCycleLR(
    optimizer, max_lr=1e-2, steps_per_epoch=len(train_loader), epochs=10
)
Adam vs AdamW Adam applies weight decay incorrectly (adds it to the gradient, so it interacts with the adaptive learning rate). AdamW decouples weight decay from the gradient update. For transformers especially, always prefer AdamW.

Eval Mode & no_grad

model.eval()    # disables Dropout, fixes BatchNorm running stats

with torch.no_grad():
    for X, y in val_loader:
        logits = model(X.to(device))
        preds = logits.argmax(dim=-1)
model.eval() and no_grad() are independent eval() changes layer behavior. no_grad() stops gradient tracking for memory/speed. You almost always want both during validation. Forgetting eval() means Dropout is still active and BatchNorm uses batch stats — your validation metrics will be silently wrong.

torch.inference_mode — the faster no_grad

# Newer alternative to no_grad — more aggressive optimizations
# Tensors created inside cannot be used in autograd later
with torch.inference_mode():
    preds = model(X)   # use this for inference/deployment

# Use no_grad when you might need gradients from the result later
# Use inference_mode for pure evaluation — it's faster

Metrics

Computing metrics correctly requires accumulating over the full validation set, not averaging batch averages (which are biased when batch sizes differ).

Classification: accuracy, top-k

model.eval()
all_preds, all_labels = [], []

with torch.inference_mode():
    for X, y in val_loader:
        logits = model(X.to(device))
        all_preds.append(logits.cpu())
        all_labels.append(y)

all_preds  = torch.cat(all_preds)    # (N, num_classes)
all_labels = torch.cat(all_labels)   # (N,)

# Top-1 accuracy
acc = (all_preds.argmax(1) == all_labels).float().mean().item()

# Top-5 accuracy
top5 = all_preds.topk(5, dim=1).indices   # (N, 5)
top5_acc = (top5 == all_labels.unsqueeze(1)).any(dim=1).float().mean().item()

Classification: precision, recall, F1 (manual)

# Binary classification
preds  = (torch.sigmoid(logits) > 0.5).long()
labels = y

TP = ((preds == 1) & (labels == 1)).sum().float()
FP = ((preds == 1) & (labels == 0)).sum().float()
FN = ((preds == 0) & (labels == 1)).sum().float()

precision = TP / (TP + FP + 1e-8)
recall    = TP / (TP + FN + 1e-8)
f1        = 2 * precision * recall / (precision + recall + 1e-8)

Using torchmetrics (recommended for production)

# pip install torchmetrics
from torchmetrics import Accuracy, F1Score, ConfusionMatrix

acc_metric = Accuracy(task='multiclass', num_classes=10).to(device)

model.eval()

    for X, y in val_loader:
        X, y = X.to(device), y.to(device)
        preds = model(X)

        f1_metric.update(preds, y)

print(acc_metric.compute())   # final value over full val set
print(f1_metric.compute())
acc_metric.reset()            # clear state for next epoch

Regression metrics

preds  = model(X.to(device)).squeeze()

Validation loop — full template

def evaluate(model, loader, criterion, device):
    model.eval()
    total_loss = 0.0
    correct    = 0
    n          = 0

    with torch.inference_mode():
        for X, y in loader:
            X, y = X.to(device), y.to(device)
            logits = model(X)
            loss   = criterion(logits, y)

            total_loss += loss.item() * X.size(0)  # weight by batch size
            correct    += (logits.argmax(1) == y).sum().item()
            n          += X.size(0)

    return {
        'loss': total_loss / n,
        'acc':  correct / n
    }

# In training loop:
val_metrics = evaluate(model, val_loader, criterion, device)
print(f"val_loss={val_metrics['loss']:.4f}  val_acc={val_metrics['acc']:.4f}")

# CrossEntropyLoss: takes raw logits (NOT softmax output)
criterion = nn.CrossEntropyLoss()
loss = criterion(logits, labels)   # logits: (B,C) float, labels: (B,) long

# Label smoothing — prevents overconfidence
criterion = nn.CrossEntropyLoss(label_smoothing=0.1)

# Class weights — for imbalanced datasets
weights = torch.tensor([1.0, 5.0, 2.0])   # class 1 is rare, upweight it
criterion = nn.CrossEntropyLoss(weight=weights.to(device))

# BCEWithLogitsLoss: binary or multi-label
criterion = nn.BCEWithLogitsLoss()
loss = criterion(logits, labels.float())  # labels must be float [0.0 or 1.0]

# Positive weight for imbalanced binary classification
pos_weight = torch.tensor([9.0])   # 1:9 negative:positive ratio
criterion  = nn.BCEWithLogitsLoss(pos_weight=pos_weight.to(device))

# Custom loss — just a function of tensors
def focal_loss(logits, labels, gamma=2.0):
    bce  = F.binary_cross_entropy_with_logits(logits, labels, reduction='none')
    prob = torch.sigmoid(logits)
    pt   = torch.where(labels == 1, prob, 1 - prob)
    return (((1 - pt) ** gamma) * bce).mean()
reduction parameter All loss functions accept reduction='mean' (default), 'sum', or 'none'. Use 'none' when you need per-sample losses (e.g. to apply sample weights manually, or to compute focal loss). Use 'sum' carefully — it scales with batch size, which can destabilize training if your batch size varies.

Inference Patterns

Probabilities and predictions from logits

model.eval()
with torch.inference_mode():
    logits = model(X.to(device))          # raw scores (B, C)

# Multi-class
probs = torch.softmax(logits, dim=-1)    # probabilities, sum to 1
pred_class = logits.argmax(dim=-1)       # class index (B,)
conf, pred = probs.max(dim=-1)           # confidence + class together

# Binary
prob_pos = torch.sigmoid(logits)          # P(positive)
pred_bin = (prob_pos > 0.5).long()

# Top-k predictions with confidence
top_confs, top_classes = probs.topk(3, dim=-1)  # top 3 per sample

Batched inference on a raw dataset

def predict(model, data_list, batch_size, device):
    """Run inference on a list of inputs, return all predictions."""
    model.eval()
    all_preds = []
    loader = DataLoader(data_list, batch_size=batch_size)

    with torch.inference_mode():
        for batch in loader:
            batch = batch.to(device)
            logits = model(batch)
            all_preds.append(logits.cpu())

    return torch.cat(all_preds)

Test-time augmentation (TTA)

# Average predictions over multiple augmented versions of the same input
def tta_predict(model, x, augmentations, device):
    model.eval()
    preds = []
    with torch.inference_mode():
        for aug in augmentations:
            x_aug = aug(x).to(device)
            preds.append(torch.softmax(model(x_aug), dim=-1))
    return torch.stack(preds).mean(0)   # ensemble average

Device mismatch error The most common PyTorch runtime error. Model on CUDA, input on CPU (or vice versa) → crash. Always move both model and batch to the same device inside the training loop.
