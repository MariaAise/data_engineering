01 SELECT Clause Tricks
CASE WHEN / Expressions
SELECT
  o.order_id,
  o.total,
  CASE
    WHEN o.total >= 1000 THEN 'VIP'
    WHEN o.total >= 500  THEN 'Standard'
    ELSE 'Basic'
  END AS tier,
  o.total * 0.9 AS discounted,
  CONCAT(c.first_name, ' ', c.last_name) AS full_name
FROM orders o
JOIN customers c ON o.customer_id = c.id;
DISTINCT & Aggregates
SELECT
  COUNT(*)                    AS total_rows,
  COUNT(DISTINCT customer_id) AS unique_customers,
  SUM(total)                  AS revenue,
  AVG(total)                  AS avg_order,
  MIN(created_at)             AS first_order,
  MAX(created_at)             AS last_order
FROM orders;
02 WHERE vs HAVING 
-- WHERE filters rows BEFORE grouping
SELECT department, COUNT(*) AS headcount
FROM employees
WHERE status = 'active'        -- filter rows first
GROUP BY department
HAVING COUNT(*) > 5            -- then filter groups
ORDER BY headcount DESC;
ClauseFiltersCan use aggregate?
WHEREIndividual rowsNo
HAVINGGrouped resultsYes (COUNT, SUM…)
03 JOINs
INNER JOIN

Only rows where the condition matches in BOTH tables.

LEFT JOIN

All rows from left table + matching rows from right. NULLs where no match.

RIGHT JOIN

All rows from right table + matching rows from left. Rarely preferred over LEFT.

FULL OUTER JOIN

MySQL doesn't support natively — emulate with LEFT + UNION + RIGHT.

Multiple JOINs + Self-Join
-- Multiple JOINs
SELECT o.id, c.name, p.title, oi.qty
FROM orders o
JOIN customers   c  ON o.customer_id = c.id
JOIN order_items oi ON o.id          = oi.order_id
JOIN products    p  ON oi.product_id = p.id;

-- Self-join: find employees and their manager
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- FULL OUTER JOIN emulation in MySQL
SELECT * FROM a LEFT JOIN  b ON a.id = b.a_id
UNION
SELECT * FROM a RIGHT JOIN b ON a.id = b.a_id;
04 Subqueries
In WHERE / IN SELECT / Correlated
-- Subquery in WHERE
SELECT name FROM products
WHERE price > (SELECT AVG(price) FROM products);

-- Subquery in FROM (derived table)
SELECT dept, AVG(salary) AS avg_sal
FROM (
  SELECT department AS dept, salary
  FROM employees
  WHERE status = 'active'
) active_emp
GROUP BY dept;

-- Correlated subquery (runs per row)
SELECT name, salary
FROM employees e
WHERE salary > (
  SELECT AVG(salary)
  FROM employees
  WHERE department = e.department -- correlated
);
Tip: Correlated subqueries run once per row. For large tables, rewrite using a JOIN or CTE for better performance.
05 CTEs (WITH clause)
Simple CTE + Chained CTEs
-- Simple CTE
WITH monthly_revenue AS (
  SELECT
    DATE_FORMAT(created_at, '%Y-%m') AS month,
    SUM(total) AS revenue
  FROM orders
  GROUP BY month
)
SELECT month, revenue,
  revenue - LAG(revenue) OVER (ORDER BY month) AS delta
FROM monthly_revenue;

-- Chained CTEs
WITH
active_users AS (
  SELECT id FROM users WHERE active = 1
),
their_orders AS (
  SELECT o.* FROM orders o
  JOIN active_users u ON o.user_id = u.id
)
SELECT COUNT(*), SUM(total) FROM their_orders;
Recursive CTE (MySQL 8+)
WITH RECURSIVE org_tree AS (
  -- Anchor: start at the top
  SELECT id, name, manager_id, 0 AS depth
  FROM employees WHERE manager_id IS NULL
  UNION ALL
  -- Recursive: walk down
  SELECT e.id, e.name, e.manager_id, t.depth + 1
  FROM employees e
  JOIN org_tree t ON e.manager_id = t.id
)
SELECT * FROM org_tree;
06 Window Functions (MySQL 8+)
OVER() anatomy
FUNCTION() OVER (
  PARTITION BY column      -- like GROUP BY, but keeps all rows
  ORDER BY     column      -- defines row order within partition
  ROWS BETWEEN 1 PRECEDING AND CURRENT ROW  -- frame spec
)
Ranking Functions
SELECT
  name, department, salary,
  ROW_NUMBER()   OVER (PARTITION BY department ORDER BY salary DESC) AS row_num,
  RANK()         OVER (PARTITION BY department ORDER BY salary DESC) AS rank,
  DENSE_RANK()   OVER (PARTITION BY department ORDER BY salary DESC) AS dense_rank,
  NTILE(4)       OVER (ORDER BY salary DESC)                         AS quartile
FROM employees;

-- RANK vs DENSE_RANK: salaries 100, 100, 90
-- RANK:       1, 1, 3   (gap after tie)
-- DENSE_RANK: 1, 1, 2   (no gap)
Running Totals & Navigation
SELECT
  order_date, total,
  SUM(total)  OVER (ORDER BY order_date)   AS running_total,
  AVG(total)  OVER (ORDER BY order_date
    ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS moving_avg_7d,
  LAG(total, 1)  OVER (ORDER BY order_date) AS prev_order,
  LEAD(total, 1) OVER (ORDER BY order_date) AS next_order
FROM orders;
07 Set Operations 
-- UNION: combine, remove duplicates
SELECT email FROM customers
UNION
SELECT email FROM suppliers;

-- UNION ALL: keep duplicates (faster)
SELECT id, total FROM orders_2023
UNION ALL
SELECT id, total FROM orders_2024;
MySQL 8.0.31+ supports INTERSECT and EXCEPT natively. For older versions, emulate with EXISTS / NOT EXISTS or LEFT JOIN.
-- INTERSECT emulation (pre-8.0.31)
SELECT email FROM customers
WHERE email IN (SELECT email FROM newsletter);

-- EXCEPT emulation
SELECT email FROM customers
WHERE email NOT IN (SELECT email FROM unsubscribes);
08 Filtering & Conditions 
-- IN, NOT IN
SELECT * FROM orders
WHERE status IN ('shipped', 'delivered');

-- EXISTS (faster than IN for large sets)
SELECT * FROM customers c
WHERE EXISTS (
  SELECT 1 FROM orders o
  WHERE o.customer_id = c.id
);

-- BETWEEN (inclusive)
WHERE created_at BETWEEN '2024-01-01' AND '2024-12-31'

-- LIKE patterns
WHERE email LIKE '%@gmail.com'    -- ends with
WHERE name  LIKE 'John%'          -- starts with
WHERE code  LIKE 'A_C'            -- single char wildcard

-- NULL checks
WHERE deleted_at IS NULL
WHERE deleted_at IS NOT NULL
OperatorUse Case
IN (list)Match any value in a list
EXISTSCheck row existence (preferred for subqueries)
ANY / ALLCompare vs each result of subquery

10 Query Execution Order

MySQL processes clauses in this logical order — not the order you write them. This is why you can't reference a SELECT alias in WHERE.

1

2
JOIN
3

4
GROUP BY
5

6

7

8

9

Why this matters: Aliases defined in SELECT don't exist yet when WHERE runs. Use a CTE or subquery to filter on a computed alias.
11 Common Query Patterns
Top-N Per Group
-- Top 3 earners per department
WITH ranked AS (
  SELECT name, department, salary,
    ROW_NUMBER() OVER (
      PARTITION BY department
      ORDER BY salary DESC
    ) AS rn
  FROM employees
)
SELECT * FROM ranked WHERE rn <= 3;
Pivot (CASE WHEN aggregation)
SELECT
  product_id,
  SUM(CASE WHEN MONTH(sale_date) = 1 THEN qty ELSE 0 END) AS jan,
  SUM(CASE WHEN MONTH(sale_date) = 2 THEN qty ELSE 0 END) AS feb,
  SUM(CASE WHEN MONTH(sale_date) = 3 THEN qty ELSE 0 END) AS mar
FROM sales
GROUP BY product_id;
Deduplication
-- Keep only the latest row per user
WITH deduped AS (
  SELECT *,
    ROW_NUMBER() OVER (
      PARTITION BY user_id
      ORDER BY created_at DESC
    ) AS rn
  FROM events
)
SELECT * FROM deduped WHERE rn = 1;
Running Total / Cumulative Sum
SELECT
  sale_date,
  amount,
  SUM(amount) OVER (
    ORDER BY sale_date
    ROWS BETWEEN UNBOUNDED PRECEDING
            AND CURRENT ROW
  ) AS cumulative
FROM sales;
12 Performance & Optimization Tips
SARGable vs Non-SARGable
-- ❌ NOT SARGable — wraps column in function
WHERE YEAR(created_at) = 2024
WHERE LOWER(email) = 'test@example.com'

-- ✅ SARGable — index can be used
WHERE created_at BETWEEN '2024-01-01' AND '2024-12-31'
WHERE email = 'test@example.com'  -- use CI collation
EXPLAIN
EXPLAIN SELECT * FROM orders
WHERE customer_id = 42;

-- Look for: type = 'ALL' (full scan = bad)
-- Good types: ref, eq_ref, range, const
-- Check: key (index used), rows (estimated)

EXPLAIN ANALYZE SELECT ...  -- MySQL 8+ actual stats
PracticeReason
Avoid SELECT *Fetches unneeded columns, can't use covering index
EXISTS over INStops scanning on first match for large subqueries
UNION ALL over UNIONSkips dedup step when you know results are distinct
Limit subquery scopeFilter early with WHERE before JOINing
Use covering indexesIndex includes all columns the query needs

Index Hints
-- Force an index
SELECT * FROM orders
FORCE INDEX (idx_customer_id)
WHERE customer_id = 5;

-- Ignore an index
SELECT * FROM orders
IGNORE INDEX (idx_created_at)
WHERE status = 'pending';
